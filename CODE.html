1. 文件树（规划）

breadknife/               # 项目群根（单一二进制 bread：daemon / 窗口 / 客户端 三角色）
├── breadpan/             # 成员项目（内部结构见其 CODE.html）
└── src/
    ├── main.rs           # 入口：子命令分发（daemon / open / close / …）
    ├── daemon/           # bread daemon（Model，无 GTK）
    │   ├── mod.rs        # 装配与事件循环；停机
    │   ├── hyprland.rs   # Hyprland IPC 只读监听：显示器 / 工作区 / 活动窗口
    │   ├── tray.rs       # breadpan 组装（Tray::start）+ 事件泵 + 交互执行 + 菜单读取
    │   ├── system.rs     # UPower / NetworkManager / MPRIS；pamixer / brightnessctl / ddcutil
    │   ├── notify.rs     # org.freedesktop.Notifications 实现（通知守护角色）
    │   ├── windows.rs    # 窗口进程生命周期（spawn / kill / 崩溃重启）；显示器 → 窗口映射
    │   └── ipc.rs        # Unix socket 服务端：请求应答 + 状态推送
    ├── window/           # 窗口进程（View，GTK4）
    │   ├── mod.rs        # 装配与 GTK 主循环；层壳窗口实例；数据分发（喂给对应模块）
    │   ├── ipc.rs        # IPC 客户端：快照 / 推送订阅、请求投递
    │   ├── style.rs      # 按 daemon 跟踪的路径加载 CSS；reload 重读
    │   └── modules/
    │       ├── workspace.rs    # 三态工作区条（渲染 + 手势捕获）
    │       ├── window_info.rs  # 活动窗口图标 / 标题 + 通知中心弹窗
    │       ├── tray.rs         # 托盘模块：图标渲染直出 + 手势捕获 + 菜单弹窗（自绘：懒加载 / 激活 / 刷新）
    │       └── center.rs       # 控制中心入口与弹出面板
    ├── shared/           # daemon / 窗口共用
    │   ├── protocol.rs   # IPC 消息类型（请求 / 应答 / 推送）
    │   └── model.rs      # 共享数据模型（窗口切片、状态增量）
    └── assets/
        └── bread.css     # 默认主题（变量定义）
  • 同一二进制承载三种角色：daemon、窗口（由 daemon spawn，bread window --monitor …）、客户端（指令转发）；
  • daemon/ 无 GTK 依赖；GTK 与 breadpan 渲染直出只在 window/；
  • 渲染归模块：窗口主进程只维护窗口实例与数据分发，一切渲染（含托盘菜单弹窗）由对应模块包干。

// shared/protocol.rs —— daemon ↔ 窗口 IPC（请求均有应答；推送无应答）
pub enum Request {                       // 窗口 → daemon
    TrayMenuRead { item: TrayItemId },
    TrayMenuExpand { item: TrayItemId, entry: i32 },
    TrayInteract { item: TrayItemId, input: InteractInput },   // left / middle / right / scroll_*（含坐标、步数）
    TrayMenuActivate { item: TrayItemId, entry: i32 },
    WorkspaceSwitch { id: u32 },
    MediaControl { op: MediaOp },
    VolumeSet { value: f64 },
    BrightnessSet { value: f64 },
}

pub enum Response {
    Menu(MenuSnapshot),                  // 菜单读取
    Outcome(InteractionOutcome),         // 交互路由结果
    Ack,                                 // 受理
}

pub enum Push {                          // daemon → 窗口
    Snapshot(WindowSnapshot),            // 首连全量
    State(StateDelta),                   // 工作区 / 活动窗口 / 系统状态增量
    TrayState(TrayUpdate),               // 托盘项 增 / 删 / 改（携带 Item 数据）
    MenuChanged { item: TrayItemId },    // 定向：弹窗打开期间的刷新触发
    Notification(NotificationRecord),
    Reload,                              // 按 daemon 跟踪的路径重读 CSS
    Visibility(bool),
}

// shared/model.rs —— 跨进程共享数据
pub struct WindowSnapshot {
    monitor: String,
    workspaces: Vec<WorkspaceState>,
    active: Option<ActiveWindow>,
    tray: Vec<TrayItem>,                 // breadpan 数据模型
    notifications: Vec<NotificationRecord>,
    css_path: String,
}
pub struct WorkspaceState { id: u32, phase: WorkspacePhase }    // 激活 / 空置 / 紧急
pub struct ActiveWindow { class: String, title: String }
pub struct TrayUpdate;                // 托盘项增删改 + breadpan TrayItem / IconSource
pub struct NotificationRecord { id: u32, app_name: String, app_icon: String, summary: String, body: String }

// daemon/hyprland.rs
pub struct MonitorId(String);
pub enum HyprlandEvent {
    MonitorAdded(MonitorId),
    MonitorRemoved(MonitorId),
    Workspace(String),          // "workspace>>X"
    ActiveWindow { class: String, title: String },
    ConfigReloaded,
}

// daemon/mod.rs
pub struct Daemon { /* Hyprland 连接、breadpan 句柄、系统服务、窗口表、IPC 服务端 */ }
impl Daemon {
    pub fn start() -> Result<Self, Error>;
    pub fn run(self) -> Result<(), Error>;              // 事件循环（Hyprland / DBus / 托盘 / IPC）
    pub fn on_control(&mut self, cmd: ControlCommand);
    fn on_hyprland_event(&mut self, ev: HyprlandEvent);
    fn spawn_window(&mut self, monitor: &str);
    fn kill_window(&mut self, monitor: &str);
    fn push(&self, window: WindowId, msg: Push);        // 定向推送
    fn broadcast(&self, msg: Push);                     // 全窗口推送
}

pub enum ControlCommand {                                // bread.sock 行协议（ARCHITECTURE §4.2）
    Open { name: String, monitor: Option<String> },
    Close { name: String, monitor: Option<String> },
    OpenAll,
    CloseAll,
    Exit,
    Reload,
    Toggle,
}

    fn request(&self, req: Request) -> Result<Response, Error>;
}
