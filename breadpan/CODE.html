1. 文件树（规划）

src/
├── lib.rs               # 公开面出口（核心 + gtk 整合层）
├── handle.rs            # TrayConfig + TrayHandle 前端（指令受理 → 内部线程投递）
├── events.rs            # TrayEvent、单事件通道（tokio mpsc；start 交付唯一 Receiver）
├── tracker.rs           # Item 登记簿记（最小集）：报到 / 反向发现 / 死亡清理
├── props.rs             # 属性即时读取（zvariant/serde → 原生结构）
├── dbus/
│   ├── mod.rs           # DBus 域出口（内部聚合，不直接对外）
│   ├── bus.rs           # 会话总线连接（zbus tokio 特性）与库内 tokio 运行时宿主
│   ├── watcher.rs       # 登记中心服务：名持有（不可替换）、所有权丢失上报、停机
│   ├── iface.rs         # org.kde.StatusNotifierWatcher 协议面（ARCHITECTURE §3.4）
│   ├── heartbeat.rs     # 周期存活探测：结果写入登记簿记（alive 字段）
│   ├── menu.rs          # dbusmenu 现场读取（零缓存·逐层懒加载）
│   └── interact.rs      # 语义输入路由（ARCHITECTURE §2.7）+ InteractionResult 回执
└── gtk/
    ├── mod.rs           # 整合层出口（渲染直出；主线程约束说明）
    └── paintable.rs     # 渲染直出：主题命中 → 主题图标；未命中 → 位图纹理

examples/                # 唯一示例目录：可运行的用法实例（README「实例代码」表）
├── minimal.rs           # 最小消费方：start → 消费事件流 → 打印快照与事件
└── interactions.rs      # 交互式宿主：事件消费线程维护卡片册 + REPL
                          （list / info / menu·expand / click·middle·right / scroll / mclick / alive）

分组约定：dbus/ 收拢 DBus 域实现——总线连接、Watcher 服务与协议面、存活探测、 dbusmenu 读取、交互路由；登记簿记与属性取数（tracker、props）留顶层， handle/events 为公开前端。

运行约束：内部线程固定为 1——宿主 tokio CurrentThread 运行时，DBus 收发（zbus tokio 特性）、 事件组装与投递全部在此异步线程完成，不创建线程池；事件通道为单实例 tokio mpsc （start 交付唯一 Receiver；消费端与执行器无关），消费端由消费方经 glib::spawn_future_local 驱动， 库不创建 GLib 上下文。

2. 内部类型与方法签名

设计态骨架：描述各模块的内部组装方式与职责边界，实现落地后以实际代码为准。 公开类型与签名（Tray / TrayHandle / TrayEvent / 数据模型 / TrayError / gtk）见 ARCHITECTURE.html §3，此处不重复。

// ── handle.rs：TrayConfig + TrayHandle 前端 = Arc<RuntimeShared> 的薄封装 ──
//    （TrayConfig 公开定义见 ARCHITECTURE §3，与前端同文件承载，此处不重复）
pub(crate) struct RuntimeShared {
    watcher: WatcherService,        // 登记中心服务；所有权丢失 → on_lost → Displaced 终局
    tracker: RwLock<ItemTracker>,
    events: Arc<EventQueue>,
    pending: HashSet<TrayItemId>, // 变更合并待通告集合（仅内部线程触及）：信号 → insert；节拍（debounce_ms）到期 drain → Changed(id)
    bus: BusCenter,               // 同步方法经 rt.block_on 受理指令（受 timeout_ms 约束）
    stopped: AtomicBool,
}

// ── events.rs：事件队列（单实例单通道；start 交付唯一 Receiver，无追加订阅）──
pub(crate) struct EventQueue {
    tx: tokio::sync::mpsc::Sender<TrayEvent>,   // 唯一通道发送端；Receiver 仅经 Tray::start 交付
    dropped: AtomicU64,                       // 背压丢弃告警计数（单通道）
}
impl EventQueue {
    /// 建队：(队列, 唯一 Receiver)；容量为库内定值，此后库内不再发放 Receiver
    pub(crate) fn new() -> (Self, tokio::sync::mpsc::Receiver<TrayEvent>);
    /// try_send：满则丢弃该事件并递增 dropped（有损交付，不阻塞内部线程）
    pub(crate) fn send(&self, ev: TrayEvent);
    /// 关闭通道：消费端 recv() 此后返回 None（停机路径）
    pub(crate) fn shutdown(&self);
}

// ── tracker.rs：登记簿记（最小集：唯一名 + 对象路径 + 存活 + 路由标记；详情一律现场读取）──
pub(crate) struct ItemEntry {
    service: String,   // 总线唯一名
    path: String,      // Item 对象路径（交互寻址用）
    alive: bool,       // 最近一次存活探测结论（报到时置 true，此后由周期探测覆盖）
    menu: Option<String>, // dbusmenu 对象路径（注册时探测 Menu 属性；None = 无 DBusMenu，右键 fallback）
    menu_only: bool,   // 整项即菜单（左键路由依据；注册时探测，随变更信号就地更新）
}
pub(crate) struct ItemTracker { items: HashMap<TrayItemId, ItemEntry> }
impl ItemTracker {
    /// 启动期反向发现：ListNames 命中 org.kde.StatusNotifierItem- 前缀 → Introspect 探测 → 最小集入册
    pub(crate) fn discover_existing(&mut self) -> Vec<TrayItemId>;
    /// 报到受理：最小集幂等插入（含 Menu / ItemIsMenu 属性探测）；已存在 → Ok(false)
    pub(crate) fn register(&mut self, bus: &BusCenter, service: &str, path: &str) -> Result<bool, TrayError>;
    /// 死亡清理：移除该唯一名全部条目，返回被移除者
    pub(crate) fn unregister(&mut self, service: &str) -> Vec<TrayItemId>;
    /// 存活结论（登记簿记的 alive 字段，零 DBus 流量）；不在册 → None
    pub(crate) fn is_alive(&self, id: &TrayItemId) -> Option<bool>;
    /// 交互寻址：对象路径（登记最小集内的信息）
    pub(crate) fn path_of(&self, id: &TrayItemId) -> Option<&str>;
}

// ── props.rs：属性即时读取 ──
/// 现场读取单个 Item 全部官方属性，经 zvariant/serde 反序列化为 TrayItem
/// （ARCHITECTURE §3.3 字段表）；items()/item()/Snapshot 的统一取数路径
pub(crate) fn fetch_item(bus: &BusCenter, entry: &ItemEntry) -> Result<TrayItem, TrayError>;

// ── dbus/bus.rs：会话总线连接与 tokio 运行时宿主 ──
pub(crate) struct BusCenter {
    rt: tokio::runtime::Runtime,              // 库内私有 CurrentThread 运行时（独占内部线程）
    conn: zbus::Connection,                   // 启用 zbus tokio 特性，DBus 收发全异步
}
impl BusCenter {
    pub(crate) fn connect(timeout: Duration) -> Result<Self, TrayError>;
    /// 注册本端服务名（如 Host 服务名）；name_owner_changed 回调供跟踪方使用
    pub(crate) fn register_name(&self, name: &str) -> Result<(), TrayError>;
    pub(crate) fn watch_owner(&self, name: &str, on_change: impl Fn(Option<String>) + Send + 'static);
    pub(crate) fn call(&self, dest: &str, path: &str, iface: &str, method: &str,
                       body: &[zbus::zvariant::Value]) -> Result<Vec<zbus::zvariant::OwnedValue>, TrayError>;
}

// ── dbus/watcher.rs：登记中心服务（名持有 / 所有权丢失上报 / 停机）──
pub(crate) struct WatcherService;
impl WatcherService {
    /// 申请 Watcher 名（不声明 allow-replacement，持有期不可被替换）并挂载协议面；

    /// 释放名字、卸载协议面（shutdown 路径）；幂等
    pub(crate) fn stop(&self);
}

3. 关键流程调用栈（伪码）

3.1. 启动（start）

流程 Tray::start(config)
├── 连接会话总线
├── 启动内部 tokio 运行时（CurrentThread，独占内部线程）
├── WatcherService::start
│    ├── 申请 Watcher 名（不可被替换）
│    │    └── 已被他人持有 → 返回 NameTaken（多 Watcher 不能共存）
│    └── 挂载协议面（§2 iface，ARCHITECTURE §3.4）
├── 反向发现存量 Item（ItemTracker::discover_existing）
│    └── ListNames 命中官方命名模式 → Introspect 探测 → 最小集入册
├── 订阅 NameOwnerChanged（死亡清理，§3.3）
├── 逐个现场读取（fetch_item）→ 事件入队 Snapshot(全部 Item 记录)
└── 返回 (Handle, 事件通道)

3.2. 变更通告合并（固定窗口节流）

内部 tokio 运行时: 收到 PropertiesChanged(item)
             或官方 New* 信号 (NewStatus/NewIcon/NewTitle/
                               NewToolTip/NewOverlayIcon/NewAttentionIcon)
├── pending.insert(item)（仅记入待通告集合；无计时重置——连发不推迟交付）
└── 固定节拍到期（每 debounce_ms 一次，与信号无关）
     └── pending.drain() → 逐个 Changed(id) → 入队 → 投递（单通道，满则丢弃）

合并针对的是重读放大：Item 侧的双重上报（New* 与 PropertiesChanged 并发、状态切换伴随连发） 与进度型 Item 的高频刷新，经零缓存架构放大为消费方的重复现场读取；窗口节流把重读频率压到 每 Item 至多 1 次/窗口。事实事件幂等，窗口内折叠不丢信息。

3.3. 报到与死亡清理

内部: 收到 RegisterStatusNotifierItem(唯一名+路径)
├── ItemTracker::register（最小集入册；幂等，已存在则忽略）
│    └── 新登记 → 订阅其 PropertiesChanged 与官方 New* 信号（SNI §3.3）
├── 广播 StatusNotifierItemRegistered（协议面保真，§2 iface）
└── 入队 Added(id)（数据由消费方现场读取）

内部: 收到 NameOwnerChanged(唯一名, 旧属主 → "")
├── ItemTracker::unregister → 被移除条目
├── 逐条广播 StatusNotifierItemUnregistered
└── 逐条入队 Removed

3.4. 菜单现场读取（Menu 流程，interact 与 menu()/menu_expand() 共用）

流程 menu(id) / menu_expand(id, entry_id)
├── 校验 Item 在册（否则 ItemNotFound）
├── dbusmenu 对象路径取自登记簿记（注册时探测 Menu 属性；None = 无菜单，不可拉取）
├── AboutToShow(parent)（常规前置；应答异常视为未变更；根层 parent=0，展开层=entry_id）
├── GetLayout(parentId=parent, recursionDepth=1)（懒加载：每次一层）
├── GetGroupProperties
├── 组装该层 MenuSnapshot
│    ├── 解析助记符 "_" → label
│    ├── 保留原文 → raw_label
│    └── 按 visible/enabled/children-display/toggle-type/toggle-state 归一化条目
└── 返回单层快照（不落任何缓存；深层经 menu_expand 按需展开）

内部: 收到 LayoutUpdated / ItemsPropertiesUpdated
└── 入队 MenuChanged(item)（不代为重拉）

3.5. 输入路由与回执

流程 left_click(id) / middle_click(id) / right_click(id) / scroll_up(id) 等
   （每个语义输入一个公开方法；缺省 x/y=(0,0)、steps=1，_at/_by 变体传显式值）
├── 校验 Item 在册（否则 ItemNotFound）
├── 路由（§2.7 表，依据登记簿记 menu_only / menu，零查询）
│    ├── left_click → menu_only 且 DBusMenu 在册 → Menu 流程（§3.4）；否则 Activate(x,y)
│    ├── middle_click → SecondaryActivate(x,y)
│    ├── right_click → DBusMenu 在册 → Menu 流程（§3.4）；否则 ContextMenu(x,y)
│    └── scroll_up/down/left/right → Scroll(∓steps, vertical / horizontal)
├── Menu 流程 → 同步返回 InteractionOutcome::Menu(快照)
├── VOID 调用（受理即返回）→ 同步返回 InteractionOutcome::Dispatched
└── 调用结果回内部线程
     └── 入队 InteractionResult{op, 成功/失败}

3.6. 存活探测循环

内部 tokio 运行时: 周期到期
├── 对每个在册 Item 做轻量属性读取（超时 timeout_ms）
├── 成功 → alive = true；失败 → alive = false（覆盖最近一次结果）
└── 结果仅写入登记簿记，不产生事件——消费方经 is_alive(id)/item(id) 读取

3.7. 所有权丢失与停机

内部: 收到 NameLost（极罕见边缘路径：持有期外部顶替一律失败，
      仅总线层异常——如总线守护重启——可触发）
├── 清空 ItemTracker 全部簿记
├── 服务转停止形态（不再受理登记）
└── 入队 Displaced → 停机（终局报错：被其它 Watcher 取代；不重试、不强制夺回）

流程 shutdown()
├── WatcherService::stop（释放 Watcher 名 → 总线上角色下线）
├── 清空登记簿记
├── 停内部 tokio 运行时
└── 置 Stopped（幂等：重复调用直接返回；最后一个 Handle 实例 drop 自动触发）
