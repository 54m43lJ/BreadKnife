1. 文件树（规划）

src/
├── lib.rs               # 公开面出口（核心 + gtk 整合层）
├── handle.rs            # TrayConfig + TrayHandle 前端（指令受理 → 内部线程投递）
├── events.rs            # TrayEvent、单事件通道（tokio mpsc；start 交付唯一 Receiver）
├── tracker.rs           # Item 登记簿记（最小集）：报到 / 反向发现 / 死亡清理
├── props.rs             # 属性即时读取（zvariant/serde → 原生结构）
├── dbus/
│   ├── mod.rs           # DBus 域出口（内部聚合）：常量、启动编排、报到受理与反向发现、全局信号订阅
│   ├── bus.rs           # 会话总线连接（zbus tokio 特性）与库内 tokio 运行时宿主
│   ├── watcher.rs       # 登记中心服务：名持有（不可替换）、所有权丢失上报、停机
│   ├── iface.rs         # org.kde.StatusNotifierWatcher 协议面（ARCHITECTURE §3.4）
│   ├── heartbeat.rs     # 周期存活探测：结果写入登记簿记（alive 字段）
│   ├── menu.rs          # dbusmenu 现场读取（零缓存·逐层懒加载）
│   └── interact.rs      # 语义输入路由（ARCHITECTURE §2.7）+ InteractionResult 回执
└── gtk/
    ├── mod.rs           # 整合层出口（渲染直出；主线程约束说明）
    └── paintable.rs     # 渲染直出：主题命中 → 主题图标；路径 → 文件纹理；未命中 → 位图纹理

examples/                # 唯一示例目录：可运行的用法实例（README「实例代码」表）
├── minimal.rs           # 最小消费方：start → 消费事件流 → 打印快照与事件
└── interactions.rs      # 交互与菜单：activate / scroll / menu()·menu_expand 现场读取 / menu_activate

分组约定：dbus/ 收拢 DBus 域实现——总线连接、Watcher 服务与协议面、存活探测、 dbusmenu 读取、交互路由；登记簿记与属性取数（tracker、props）留顶层， handle/events 为公开前端。

运行约束：内部线程固定为 1——宿主 tokio CurrentThread 运行时，DBus 收发（zbus tokio 特性）、 事件组装与投递全部在此异步线程完成，不创建线程池；事件通道为单实例 tokio mpsc （start 交付唯一 Receiver；消费端与执行器无关），消费端由消费方经 glib::spawn_future_local 驱动， 库不创建 GLib 上下文。

设计态骨架：描述各模块的内部组装方式与职责边界，实现落地后以实际代码为准。 公开类型与签名（Tray / TrayHandle / TrayEvent / 数据模型 / TrayError / gtk）见 ARCHITECTURE.html §3，此处不重复。

// ── handle.rs：TrayConfig + Tray + TrayHandle 前端 = Arc<RuntimeShared> 的薄封装 ──
//    （TrayConfig 公开定义见 ARCHITECTURE §3，与前端同文件承载，此处不重复）
pub(crate) struct RuntimeShared {
    bus: BusCenter,                       // 库内总线域句柄（Clone 廉价）
    rt_thread: thread::ThreadId,          // 内部运行时线程（停机 join 用；运行时线程自身上不得 join）
    join: Mutex<Option<thread::JoinHandle<()>>>,
}
impl RuntimeShared {
    /// 优雅停机（幂等）：set_stopped → begin_stopping → request_stop；
    /// 非 runtime 线程上 join 内部线程收尾（运行时线程自身调用时不 join——防自 join 死锁）
    fn shutdown(&self);
}
impl Tray {
    /// Tray::start（handle.rs）：建队 → BusParts::new → BusCenter::spawn_runtime
    /// → 轮询 ready 通道等 bootstrap 完成（引导总时限 max(10×timeout_ms, 10s) 守护，
    ///   超时/失败 → join 内部线程并返回 Err）→ Arc<RuntimeShared>
    pub fn start(config: TrayConfig) -> Result<(TrayHandle, Receiver<TrayEvent>), TrayError>;
}

// ── events.rs：事件队列（单实例单通道；start 交付唯一 Receiver，无追加订阅）──
pub(crate) const CHANNEL_CAPACITY: usize = 256;   // 通道容量（库内定值）
pub(crate) struct EventQueue {
    tx: tokio::sync::mpsc::Sender<TrayEvent>,   // 唯一通道发送端；Receiver 仅经 Tray::start 交付
    dropped: AtomicU64,                       // 背压丢弃告警计数（单通道）
}
impl EventQueue {
    /// 建队：(Arc<队列>, 唯一 Receiver)；此后库内不再发放 Receiver
    pub(crate) fn new() -> (Arc<Self>, tokio::sync::mpsc::Receiver<TrayEvent>);
    /// try_send：满则丢弃该事件并递增 dropped（有损交付，不阻塞内部线程）
    pub(crate) fn send(&self, ev: TrayEvent);
    /// 背压丢弃累计（Narration 观测用）
    pub(crate) fn dropped(&self) -> u64;
    /// 关闭通道：消费端 recv() 此后返回 None（停机路径）
    pub(crate) fn close(&self);
}

// ── tracker.rs：登记簿记（最小集；unique = 连接身份，同连接恒为单条目；Bus Name 报到时 id 升级）──
pub(crate) struct ItemEntry {
    service: String,   // 公共名（= TrayItemId；唯一名形态报到时暂以临时唯一名占位，
                       //  Bus Name 报到时升级并触发 NameChanged）
    unique: String,    // 属主唯一名（= 连接身份，第二轮 lookup 键；注册时 GetNameOwner 解析）
    path: String,      // Item 对象路径（交互寻址用；以最新报到为准）
    alive: bool,       // 最近一次存活探测结论（报到时置 true，此后由周期探测覆盖）
    menu: Option<String>, // dbusmenu 对象路径（注册时探测 Menu 属性；None = 无 DBusMenu，右键 fallback）
    menu_only: bool,   // 整项即菜单（左键路由依据；注册时探测，随变更信号就地更新）
}
pub(crate) struct ItemTracker { items: HashMap<TrayItemId, ItemEntry> }
impl ItemTracker {
    /// 字段匹配两轮 lookup（注册名 ↔ service、发送者唯一名 ↔ unique）：
    /// 命中原地更新，未命中插入；Bus Name 报到使临时唯一名 id 升级 → Renamed（§3.3.1）
    pub(crate) fn upsert(&mut self, entry: ItemEntry) -> Upsert;
    /// 死亡清理：按名字（服务名或属主唯一名）双条件命中，移除全部命中条目，返回被移除者
    pub(crate) fn unregister(&mut self, name: &str) -> Vec<ItemEntry>;
    /// 信号落点：(唯一名, 对象路径) 唯一命中——同连接已并入单条目，无落点歧义
    pub(crate) fn find_by_signal(&self, sender: &str, path: &str) -> Option<TrayItemId>;
    /// 就地更新路由标记（PropertiesChanged 携带 ItemIsMenu / Menu 时）
    pub(crate) fn update_routing(&mut self, service: &str, menu: Option<Option<String>>,
                                 menu_only: Option<bool>);
    /// 存活结论（登记簿记的 alive 字段，零 DBus 流量）；不在册 → None
    pub(crate) fn is_alive(&self, id: &TrayItemId) -> Option<bool>;
    /// 覆盖存活结论（heartbeat_task 周期探测回写）
    pub(crate) fn set_alive(&mut self, service: &str, alive: bool);
    /// dbusmenu 信号落点：(唯一名, 菜单对象路径) 唯一命中
    pub(crate) fn find_by_menu_signal(&self, sender: &str, path: &str) -> Option<TrayItemId>;
}

3. 关键流程调用栈（伪码）

3.1. 启动（start）

3.1.1. 总流程（bootstrap）

流程 Tray::start(config)（handle.rs）
├── EventQueue::new() → (Arc<EventQueue>, 唯一 Receiver)
├── BusParts::new(config, events)
├── BusCenter::spawn_runtime(parts)
│    └── 内部 CurrentThread 运行时线程：mod::run(bus, stop_rx, fatal_rx,
│         displaced_rx, ready_tx) → bootstrap(bus)（顺序即契约，理由见下方说明）
│         ├── watcher::mount_interfaces(bus)?（org.kde / org.freedesktop
│         │    两前缀挂同一路径 /StatusNotifierWatcher）
│         │    └── 名未在位 → 报到无从发生，快照收集不受报到竞态干扰
│         ├── spawn_signal_tasks(bus)：6 条 match rule 常驻任务
│         │    （name_owner_changed_task / item_signals_task×2 kde+fd /
│         │      properties_changed_task×2 kde+fd / dbusmenu_signals_task，
│         │      经 pump_signal_stream 订阅消费——§3.2 §3.3.2 §3.4.2）
│         ├── bus.spawn(heartbeat::heartbeat_task(bus))（周期存活探测，§3.6）
│         ├── bus.spawn(debounce_task(bus))（节流窗口，§3.2）
│         ├── discover_existing(bus)（反向发现存量 Item，§3.1.2，静默入册）
│         ├── queue_snapshot(bus)：逐个 props::fetch_item → 排序
│         │    → 事件入队 Snapshot(全部 Item 记录)
│         └── watcher::WatcherService::start(bus)?（最后申请 Watcher 名）
│              └── 已被他人持有 → 返回 NameTaken（多 Watcher 不能共存）
├── ready_rx 轮询 bootstrap 结果（引导总时限 max(10×timeout_ms, 10s) 守护）
│    └── 超时/失败 → join 内部线程并返回 Err
└── 返回 (TrayHandle{Arc<RuntimeShared>}, 事件通道 Receiver)

顺序为什么是契约：Watcher 名最后申请——申请前报到无从发生，快照投递后所有报到 以增量事件衔接（首事件即快照，无重叠无缺口）。若先申请名，等待 Watcher 出现的 Item（如 clash-verge）会在快照收集前报到，其 Added 抢在快照之前——实测复现。 订阅（spawn_signal_tasks）与周期任务（heartbeat_task / debounce_task）先于申请名同理：名在位即可能收到信号与超时，须先就位才不丢。

3.1.2. 反向发现（存量 Item 双级探测）

内部: dbus::discover_existing(bus) -> Vec<String>（过程叙述，快照投递后以 Narration 交付）
├── ListNames → 过滤 "org.kde.StatusNotifierItem-" 前缀候选
├── 逐个 probe_item_iface(bus, name) -> Result<bool, TrayError>：双级确认 SNI 接口在位
│    ├── 第一级：Introspect(name, /StatusNotifierItem) 应答 XML 含 SNI 接口名
│    │    （org.kde / org.freedesktop 任一）→ 确认
│    └── 第二级：props::read_all(name, /StatusNotifierItem) 直接属性读取
│         （Introspect 失败或应答 XML 无接口名时到达——Electron/CEF 等实现
│          不响应 Introspect，QQ 音乐实测）；Ok = 确认在位
├── 确认 → register_item(bus, name, sender=None, announce=false)
│    （静默入册：不广播、不入队 Added——快照是首事件，§3.3.1）
└── 未确认 / 探测失败 → 记入过程叙述，不中断发现循环

3.2. 变更通告合并（固定窗口节流）

内部: properties_changed_task(iface)×2 / item_signals_task(iface)×2
   （kde 与 fd 前缀各一条常驻任务；match rule 订阅经 pump_signal_stream）
├── 信号到达：落点匹配 find_by_signal(发送者唯一名, 对象路径)——
│    (唯一名, 路径) 唯一命中（双注册已并入，§3.3.1），Changed 落点确定
│    └── PropertiesChanged 另就地更新路由标记（update_routing，§3.3.1）
├── pending.insert(id)（仅记入待通告集合；无计时重置——连发不推迟交付）
└── debounce_task 固定节拍到期（每 debounce_ms 一次，与信号无关）
     └── pending.drain() → 逐个 Changed(id) → 入队 → 投递（单通道，满则丢弃）

合并针对的是重读放大：Item 侧的双重上报（New* 与 PropertiesChanged 并发、状态切换伴随连发） 与进度型 Item 的高频刷新，经零缓存架构放大为消费方的重复现场读取；窗口节流把重读频率压到 每 Item 至多 1 次/窗口。事实事件幂等，窗口内折叠不丢信息。

3.3. 报到与死亡清理

3.3.1. 报到（RegisterStatusNotifierItem 受理）

内部: 协议面收到 RegisterStatusNotifierItem(service)
      （org.kde / org.freedesktop 两前缀等价：两接口挂同一路径，§2 iface）
├── sender ← 消息头 sender（调用方唯一名，#[zbus(header)] 注入）
└── dbus::register_item(bus, service, sender, announce=true) -> Result<bool, TrayError>
     ├── split_service_arg(service, sender) -> Option<(String, String)>：注册参数三形态解析
     │    ├── "name"（规范服务名）→ (name, /StatusNotifierItem)（默认路径）
     │    ├── "name/path"（混串）→ 就地切分；path 缺前导 "/" 时补齐
     │    └── "/path"（Ayatana 风格纯对象路径）→ (sender, path)：服务名取注册方唯一名
     │         └── 无 sender（或 service 为空）→ None → TrayError::Protocol 拒绝受理
     │              （以 Narration 事件交付，不影响服务运行）
     ├── GetNameOwner(service) → 属主唯一名（失败回退 service 本身）
     ├── props::probe_routing(service, path) → (menu, menu_only)（路由标记注册时探测）
     └── ItemTracker::upsert(ItemEntry) → Upsert（字段匹配两轮 lookup，自愈并入：
          同一对象/连接不允许出现双条目）
          ├── 第一轮：注册名 == service（Bus Name 同名：应用重启竞态/同名刷新；
          │    唯一名形态注册名 == 临时条目 service）→ 命中
          ├── 第二轮：发送者唯一名 == unique（同连接先前以唯一名临时注册）→ 命中
          ├── 命中 → 原地更新（path/menu/menu_only/alive 以最新报到为准；
          │    service 仅在注册名为 Bus Name 且原为临时唯一名时升级）
          │    └── 升级（临时唯一名 → Bus Name）→ 入队 NameChanged{old, new}
          │         （同一对象 id 重键——不发 Removed/Added，
          │          是否重载数据由消费方裁断）
          ├── 两轮均未命中 → 新条目 → true
          │    └── 新登记 && announce → iface::emit_registered（StatusNotifierItemRegistered 广播）
          │         + 入队 Added(id)
          │         （数据由消费方现场读取；announce=false 仅反向发现路径——静默入册，
                     由快照承载，§3.1）

注册参数 service 与信号发送者分属两个命名体系：service 既可以是 well-known 名（规范服务名）， 也可以是唯一名——无 well-known 名的 Item 以 :1.22 形态直接报到（clash-verge 实测）。 两种形态同等入册，service 即 TrayItemId，簿记不区分。而 SNI 与 dbusmenu 信号的发送者一律是 唯一名，与注册名可能不一致：注册时经 GetNameOwner（org.freedesktop.DBus 接口的标准方法，由总线守护进程实现，本库为调用方）把注册名解析为属主唯一名，存入簿记 unique 字段；对唯一名查询返回其自身，故全部注册形态走同一代码路径。解析失败（名字无属主） 回退以 service 本身充当，不阻塞报到。该字段有两个消费点：信号归属匹配（§3.2，按发送者 唯一名 + 对象路径定位 id）与死亡清理（§3.3.2，按服务名或属主唯一名双条件命中）。

3.3.2. 死亡清理（NameOwnerChanged）

内部: name_owner_changed_task 收到 NameOwnerChanged(唯一名, 旧属主 → "")
├── ItemTracker::unregister(名字) → 被移除条目（按服务名或属主唯一名双条件命中；
│    唯一名消失 → 该连接全部条目一并下线，含双注册并入的单一 id）
├── 逐条 iface::emit_unregistered（StatusNotifierItemUnregistered 广播）
└── 逐条入队 Removed

3.4. 菜单现场读取（Menu 流程，interact 与 menu()/menu_expand() 共用）

3.4.1. 现场拉取（menu / menu_expand）

流程 menu(id) / menu_expand(id, entry_id)
   （TrayHandle::menu / menu_expand → dispatch 同步桥 → menu::read_layer）
├── 校验 Item 在册（否则 ItemNotFound）
├── dbusmenu 对象路径取自登记簿记（注册时探测 Menu 属性；None = 无菜单，不可拉取）
└── menu::read_layer(bus, entry, parent)——根层 parent=0，展开层=entry_id
     ├── AboutToShow(parent)（常规前置；应答异常视为未变更）
     ├── GetLayout(parentId=parent, recursionDepth=1)（懒加载：每次一层）
     ├── GetGroupProperties
     └── assemble → 组装该层 MenuSnapshot
          ├── 解析助记符 "_" → label
          ├── 保留原文 → raw_label
          └── 按 visible/enabled/children-display/toggle-type/toggle-state 归一化条目
          └── 返回单层快照（不落任何缓存；深层经 menu_expand 按需展开）

3.4.2. 菜单变更通告（LayoutUpdated / ItemsPropertiesUpdated）

内部: dbusmenu_signals_task 收到 LayoutUpdated / ItemsPropertiesUpdated
├── 落点匹配 find_by_menu_signal(发送者唯一名, 菜单对象路径)（唯一命中）
└── 入队 MenuChanged(item)（不代为重拉；快照投递前丢弃）

3.5. 输入路由与回执

3.5.1. 左键（同步路径与 Menu 回退）

流程 left_click(id)（TrayHandle::left_click → left_click_at(id, 0, 0)
   → input(id, Input::Left) → dispatch 同步桥（BusCenter::cmd）→ interact::route）
├── 校验 Item 在册（否则 ItemNotFound）
├── menu_only 且 DBusMenu 在册 → Menu 流程（§3.4.1）→ 同步返回 Outcome::Menu
└── Activate(x,y)（invoke_xy：就地等待往返）
     ├── 成功 → 入队 InteractionResult{Activate, 成功} → 返回 Outcome::Dispatched
     ├── 超时 → 直接报错（对端主循环失联；menu 与 Activate 同一主循环分发，
     │         菜单亦不可读——不跨前缀重试、不回退）
     ├── 失败且 DBusMenu 在册 → Menu 流程（§3.4.1）→ 返回 Outcome::Menu（无回执事件）
     │    └── 回退动机：libappindicator 系 Item 的 Activate 无处理器——XML 静态声明
     │         但分发 catch-all；不可预先探测（Introspect 说谎、探测调用会误触
     │         激活）。左键出菜单是宿主侧既定策略：菜单由宿主经 dbusmenu 渲染，
     │         不依赖 item 端激活。失败形态为同步快失败，与超时无关
     └── 失败且无菜单 → 如实报错

3.5.2. 其余语义输入（受理即返回）

流程 middle_click(id) / right_click(id) / scroll_up(id) 等
   （TrayHandle::*_click / *_by → input → dispatch 同步桥 → interact::route；
    缺省 x/y=(0,0)、steps=1，_at/_by 变体传显式值）
├── 校验 Item 在册（否则 ItemNotFound）
├── 路由（依据登记簿记 menu / menu_only，零查询）
│    ├── right_click → DBusMenu 在册 → Menu 流程（§3.4.1）→ Outcome::Menu
│    │                 └── 否则 ContextMenu(x,y)（SNI 原生 fallback）
│    ├── middle_click → SecondaryActivate(x,y)
│    └── scroll_up/down/left/right → Scroll(∓steps, vertical / horizontal)
├── 调用（invoke_xy / invoke_scroll：org.kde 前缀优先，非超时失败回退
│    org.freedesktop——兼容只导出规范原文拼写的少数实现；任一侧超时 →
│    立即报错；双前缀俱败 → 合并上报两侧错误，避免掩盖首侧真因）
├── Menu 流程 → 同步返回 Outcome::Menu；VOID 调用（受理即返回）→ 返回 Outcome::Dispatched
└── 调用结果回内部线程
     └── 入队 InteractionResult{op, 成功/失败}

3.6. 存活探测循环

内部: heartbeat_task（HEARTBEAT_PERIOD = 5s，bus.spawn 常驻）
├── GetNameOwner(自身唯一名) 探测总线（bus.conn.call_method，受 timeout_ms 约束）
│    ├── IO 类错误（InputOutput / Handshake / Connection）→ report_fatal → Fatal（终局）
│    └── 超时 → 视作繁忙，跳过本拍
├── 逐 Item（bus.rt.spawn 并发）Get(Status)（PROPS_IFACE，超时 timeout_ms）
│    ├── 成功 → ItemTracker::set_alive(service, true)
│    └── 失败 → ItemTracker::set_alive(service, false)（覆盖最近一次结果）
└── 结果仅写入登记簿记，不产生事件——消费方经 is_alive(id)/item(id) 读取

3.7. 所有权丢失与停机

3.7.1. 所有权丢失（Displaced）

内部: name_owner_changed_task 收到 NameOwnerChanged(Watcher 名, 属主 → "")
   （极罕见边缘路径：持有期外部顶替一律失败（DoNotQueue），
    仅总线层异常——如总线守护重启——可触发）
├── is_stopping / is_stopped 守卫：停机路径主动释放名引发的 NOC 在此吞掉，
│    不产生 Displaced；仅异常丢失（持有期不可被替换）才继续
├── report_displaced()（displaced watch 通道通知 mod::run）
└── run() displaced 分支：
     ├── 清空 ItemTracker 全部簿记（ItemTracker::clear）
     ├── 服务转停止形态（set_stopped，不再受理登记）
     ├── 入队 Displaced → 事件通道 close
     └── （终局报错：被其它 Watcher 取代；不重试、不强制夺回）

3.7.2. 停机（shutdown）

流程 shutdown()（Handle::shutdown / 最后一个 Handle drop → RuntimeShared::shutdown）
├── 幂等守卫（is_stopped → 直接返回）
├── set_stopped（此后 Handle 方法一律 Stopped）
├── begin_stopping（名释放引发的 NameOwnerChanged 不得触发 Displaced，§3.7.1）
├── request_stop → 内部线程 run() 的 stop 分支 → graceful_shutdown(bus)：
│    ├── watcher::release（释放 Watcher 名 → 总线上角色下线；受 timeout 约束）
│    ├── 清空登记簿记（ItemTracker::clear）
│    └── 事件通道 close（消费端 recv() → None）
└── join 内部线程收尾（运行时线程自身调用时不 join——防自 join 死锁）

3.8. 渲染直出（icon_paintable）

流程 icon_paintable(&IconSource) -> gdk::Paintable
   （GTK 主线程调用；三类失败统一回退 image-missing / 空纹理，不 panic）
├── Name(name)（主题图标名）
│    └── lookup_theme_icon(name)
│         ├── gdk::Display::default()? → IconTheme::for_display（逐显示连接取用）
│         ├── has_icon(name)（Icon Theme Spec 命中判定，含继承主题链）
│         │    ├── 否 → None → missing_icon()（image-missing 占位，
│         │    │         Icon Naming Spec 保留名，主题规范保证存在）
│         │    └── 是 → lookup_icon(name, [], LOOKUP_SIZE=48, scale=1, Ltr)
│         │              → 主题 Paintable
├── Path(path)（文件路径图标，Ayatana 生态事实扩展）
│    └── gdk::Texture::from_filename(path)
│         ├── Ok(texture) → 文件纹理
│         └── Err（文件不存在 / 读取失败 / 解码失败）→ missing_icon()
│              （与主题未命中同回退；文件为外部状态，不落任何缓存）
└── Pixmap(p)（裸位图，自带数据——DBus 读取时已拷贝入 TrayItem）
     └── pixmap_to_paintable(p)
          ├── 完整性检查：w,h > 0 且 argb.len() ≥ (w×4) × h？
          │    └── 否 → 1×1 全透明 A8r8g8b8 纹理（渲染为空，不 panic）
          └── MemoryTexture::new(w, h, A8r8g8b8, bytes, stride = w×4)
               └── 色彩对齐：SNI ARGB32 网络字节序 [A,R,G,B]、未预乘
                    ↔ gdk 非预乘 A8r8g8b8 内存格式——字节序天然对齐、
                    无需换序；预乘合成由渲染器负责
