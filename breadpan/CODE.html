1. 文件树（规划）

src/
├── lib.rs               # 公开面出口（核心 + gtk 整合层 + testing）
├── config.rs            # TrayConfig
├── handle.rs            # TrayHandle 前端（指令受理 → 内部线程投递）
├── events.rs            # TrayEvent、通道与回调分发
├── bus.rs               # 会话总线连接与 zbus 执行器宿主
├── host.rs              # Host 注册 + Item 发现/移除跟踪
├── props.rs             # Item 属性拉取 + PropertiesChanged 去抖
├── heartbeat.rs         # 周期存活探测
├── menu.rs              # dbusmenu 现场读取（零缓存）
├── interact.rs          # 交互转发 + InteractionResult 回执
├── watcher_track.rs     # Watcher 名字所有者跟踪
├── gtk/
│   ├── mod.rs           # 整合层出口（主线程约束说明）
│   ├── paintable.rs     # 渲染直出：主题命中 → 主题图标；未命中 → 位图纹理
│   └── bridge.rs        # 事件桥接：通道 → GTK 主线程（GLib 主循环）搬运
└── testing/
    ├── demo_item.rs     # 可编程演示 Item
    └── stub_watcher.rs  # 最小 Watcher 替身

2. 类型与方法签名

2.1 入口与配置

pub struct TrayConfig {
    pub timeout_ms: u64,     // 服务发现/调用/心跳超时，默认 2000 ms
    pub debounce_ms: u64,    // 属性变更合并窗口，默认 50 ms
    pub menu_max_depth: u8,  // 菜单拉取最大深度，默认 8
}

pub struct Tray;

impl Tray {
    /// 连接会话总线并启动内部线程；阻塞至完成 Host 登记（或超时）。
    /// 通道首事件为快照，与后续事件原子衔接。
    pub fn connect(config: TrayConfig)
        -> Result<(TrayHandle, Receiver<TrayEvent>), TrayError>;

    /// 底层原语（ARCHITECTURE §2 投递链 ①）：以回调方式构建。事件全部经回调投递。
    pub fn builder(config: TrayConfig) -> TrayBuilder;
}

impl TrayBuilder {
    /// 注册事件回调；可多次调用，按注册顺序串行触发
    pub fn on_event(self, cb: impl Fn(&TrayEvent) + Send + 'static) -> Self;
    /// 组装并启动，语义同 Tray::connect
    pub fn spawn(self) -> Result<TrayHandle, TrayError>;
}

2.2 Handle

#[derive(Clone)]
pub struct TrayHandle { /* Send + Sync + 'static */ }

impl TrayHandle {
    pub fn items(&self) -> Vec<TrayItem>;
    pub fn item(&self, id: &TrayItemId) -> Option<TrayItem>;
    /// 菜单现场读取：每次调用即最新全量布局（零缓存）
    pub fn menu(&self, id: &TrayItemId) -> Result<MenuSnapshot, TrayError>;

    // ── 交互转发：同步受理结论 + InteractionResult 异步回执 ──
    pub fn activate(&self, id: &TrayItemId, x: i32, y: i32) -> Result<(), TrayError>;
    pub fn secondary_activate(&self, id: &TrayItemId, x: i32, y: i32) -> Result<(), TrayError>;
    pub fn scroll(&self, id: &TrayItemId, delta: i32, orient: Orientation) -> Result<(), TrayError>;
    pub fn menu_activate(&self, id: &TrayItemId, menu_item_id: i32) -> Result<(), TrayError>;
    /// 通知 Item 某菜单即将展示（AboutToShow），返回是否建议刷新
    pub fn menu_about_to_show(&self, id: &TrayItemId, menu_item_id: i32) -> Result<bool, TrayError>;

    /// 追加订阅事件通道（独立于 on_event 回调，可多订阅者）
    pub fn subscribe(&self) -> Receiver<TrayEvent>;
    /// 优雅停机：注销 Host、停内部线程；drop 最后一个实例时自动调用
    pub fn shutdown(&self);
}

pub enum Orientation { Horizontal, Vertical }  // ← SNI Scroll(STRING orientation)：horizontal / vertical

2.3 事件

pub enum TrayEvent {
    /// 订阅起点：完整 Item 记录 + Watcher 存在性，与后续事件原子衔接
    Snapshot { items: Vec<TrayItem>, watcher_up: bool },
    Added(TrayItem),
    Removed(TrayItemId),
    /// 属性变更（去抖合并后的字段并集）
    Changed { id: TrayItemId, changed: Vec<ItemField>, item: TrayItem },
    MenuChanged(TrayItemId),
    WatcherChanged(WatcherState),
    /// 心跳结论：无响应标记置位/解除（只标记，不移除）
    Unresponsive { id: TrayItemId, unresponsive: bool },
    InteractionResult { id: TrayItemId, op: InteractionOp, result: Result<(), String> },
    /// 内部过程叙述（重连、重扫描等）
    Narration(String),
    /// 终止宣告：最后一个事件，此后 Handle 方法一律返回 Stopped
    Fatal(String),
}

pub enum WatcherState { ExternalUp, ExternalDown }
/// ← 交互指令的官方方法：SNI §3.2 Activate / SecondaryActivate / Scroll；
///    dbusmenu Event("clicked") / AboutToShow
pub enum InteractionOp { Activate, SecondaryActivate, Scroll, MenuActivate, MenuAboutToShow }
/// 变更字段并集的成员：与 SNI 官方属性组一一对应；
/// 官方信号 New*（SNI §3.3）与 PropertiesChanged 均触发相应成员入集
pub enum ItemField {
    Status,        // ← Status（官方信号 NewStatus）
    Ident,         // ← Id
    Title,         // ← Title（官方信号 NewTitle）
    Icon,          // ← IconName + IconPixmap（官方信号 NewIcon）
    OverlayIcon,   // ← OverlayIconName + OverlayIconPixmap（官方信号 NewOverlayIcon）
    AttentionIcon, // ← AttentionIconName + AttentionIconPixmap（官方信号 NewAttentionIcon）
    Tooltip,       // ← ToolTip（官方信号 NewToolTip）
    WindowId,      // ← WindowId
    Category,      // ← Category
    IsMenuOnly,    // ← ItemIsMenu
    HasMenu,       // ← Menu
}

2.4 数据模型

字段级官方映射总表见 ARCHITECTURE.html §4；此处以行内注释标注对应官方字段与语义。

/// 全局唯一键：Item 注册到 Watcher 的服务名。
/// 官方命名 org.freedesktop.StatusNotifierItem-PID-ID（SNI §3：PID=进程号，ID=同进程实例序号）；
/// Watcher 的 RegisteredStatusNotifierItems 与 StatusNotifierItemRegistered/Unregistered
/// 均以该服务名为准，对象路径不参与标识。
pub struct TrayItemId { pub service: String }

pub struct TrayItem {
    pub id: TrayItemId,                     // 唯一键（见上）
    pub category: TrayCategory,             // ← SNI Category：ApplicationStatus / Communications / SystemServices / Hardware
    pub status: TrayStatus,                 // ← SNI Status：Passive / Active / NeedsAttention
    pub ident: String,                      // ← SNI Id：应用内唯一、跨会话一致（如应用名）；识别用，不用于展示
    pub title: Option<String>,              // ← SNI Title：面向用户展示名，可比 Id 更具描述性
    pub is_menu_only: bool,                 // ← SNI ItemIsMenu（BOOL）：为真时优先展示菜单而非 Activate
    pub window_id: u32,                     // ← SNI WindowId（UINT32）；不关心时为 0
    pub icon: IconSource,                   // ← SNI IconName / IconPixmap（二选一规则见 ARCHITECTURE §4.2）
    pub overlay_icon: Option<IconSource>,   // ← SNI OverlayIconName / OverlayIconPixmap
    pub attention_icon: Option<IconSource>, // ← SNI AttentionIconName / AttentionIconPixmap
    pub tooltip: Option<ToolTip>,           // ← SNI ToolTip (STRING, a(iiay), STRING, STRING)
    pub has_menu: bool,                     // ← SNI Menu（OBJECT PATH）非空；路径由内部保存供 menu() 使用
}

/// 二选一规则固定：Name 优先，否则位图（官方建议名优先于位图，SNI §3.1.6）
pub enum IconSource { Name(String), Pixmap(Pixmap) }

/// SNI IconPixmap：签名 a(iiay)，ARGB32、网络字节序、行跨度 = width * 4（SNI §6）
pub struct Pixmap { pub width: u32, pub height: u32, pub argb: Vec<u8> }

/// ← SNI ToolTip：图标名 + 位图 + 标题 + 描述
pub struct ToolTip { pub icon: Option<IconSource>, pub title: String, pub description: String }
// description 官方语义：可含标记子集 b/i/u/a/img（SNI §7），原样透传

/// menu() 的返回值（非缓存）
pub struct MenuSnapshot { pub revision: u32, pub root: Vec<MenuItem> }
// revision ← dbusmenu GetLayout 返回的 revision（与 LayoutUpdated.revision 对应）

pub struct MenuItem {
    pub id: i32,                     // ← dbusmenu 条目数字 id（GetLayout 布局 (i,a{sv},av) 的 i）
    pub kind: MenuItemKind,          // ← type（standard/separator）+ children-display="submenu"
    pub label: String,               // ← label 已解析：__ 显示为 _，第一个剩余 _ 后字符为访问键
    pub raw_label: String,           // ← label 原始值（逃生门）
    pub enabled: bool,               // ← enabled（官方缺省 true；非默认值才上报，组装时按默认值补全）
    pub visible: bool,               // ← visible（官方缺省 true）
    pub toggled: Option<bool>,       // ← toggle-type ∈ {checkmark, radio} 时按 toggle-state：
                                     //    1→Some(true)，0→Some(false)，其余/缺省 -1→None
    pub icon_name: Option<String>,   // ← icon-name（Freedesktop 图标名）
    pub icon_data: Option<Vec<u8>>,  // ← icon-data（PNG 字节流；与 SNI 的 ARGB32 位图不同格式）
    pub children: Vec<MenuItem>,     // ← GetLayout 递归布局（recursionDepth：-1 全部/0 不递归/n 层）；
                                     //   深度截断于 menu_max_depth
}

pub enum TrayCategory { ApplicationStatus, Communications, SystemServices, Hardware } // ← Category 官方四值
pub enum TrayStatus { Passive, Active, NeedsAttention }                              // ← Status 官方三值
pub enum MenuItemKind { Standard, Separator, Submenu }

3. 关键流程调用栈（伪码）

3.1 启动（connect）

流程 connect(config)
├── 连接会话总线
├── 启动内部事件线程（zbus 执行器宿主）
├── 注册 Host 服务名（org.freedesktop.StatusNotifierHost-id，SNI §5）
├── 调用 Watcher.RegisterStatusNotifierHost
├── 探测 Watcher 存在性（IsStatusNotifierHostRegistered）→ 记录初始存亡
├── 查询在册 Item 清单（RegisteredStatusNotifierItems）
│    └── 对每个 Item
│         ├── 拉取全属性 → 建立 Item 记录
│         └── 订阅其 PropertiesChanged 与官方 New* 信号（§3.3）
├── 事件入队 Snapshot(全部 Item, Watcher 存亡)
└── 返回 (Handle, 事件通道)

3.2 属性变更去抖

内部事件线程: 收到 PropertiesChanged(item, 字段)
             或官方 New* 信号 (NewStatus/NewIcon/NewTitle/
                               NewToolTip/NewOverlayIcon/NewAttentionIcon)
├── 追加进该 item 的待发变更集
├── 刷新去抖窗口计时
└── 窗口到期
     └── 合成 Changed{字段并集, 最新记录} → 入队 → 分发给回调与全部通道

3.3 菜单现场读取

流程 menu(id)
├── 校验 Item 在册（否则 ItemNotFound）
├── GetLayout（深度截断 menu_max_depth）
├── GetGroupProperties
├── 组装 MenuSnapshot
│    ├── 解析助记符 "_" → label
│    ├── 保留原文 → raw_label
│    └── 按 visible/enabled/toggled 归一化条目
└── 返回快照（不落任何缓存）

内部: 收到 LayoutUpdated / ItemsPropertiesUpdated / AboutToShow
└── 入队 MenuChanged(item)（不代为重拉）

3.4 交互转发

流程 activate(id, x, y)   （secondary_activate / scroll / menu_activate 同构）
├── 校验 Item 在册
├── 向 Item 发起 DBus 调用（受理即返回）
├── 同步返回 Ok / Err
└── 调用结果回到内部线程
     └── 入队 InteractionResult{op, 成功/失败}

3.5 心跳循环

内部事件线程: 周期心跳
├── 对每个在册 Item 做轻量属性读取（超时 timeout_ms）
├── 成功 → 清零失败计数；若曾标记 → 入队 Unresponsive(false)
└── 失败 → 累加失败计数
     └── 达到阈值 → 入队 Unresponsive(true)（不移除）

3.6 Watcher 跟踪与停机

内部: Watcher 名字所有者变化
├── 消失 → 入队 Narration(重连准备) + WatcherChanged(ExternalDown)
│         └── 重连成功后重新发现 Item（叙述 + 修正事件）
└── 出现 → 入队 WatcherChanged(ExternalUp)

流程 shutdown()
├── 向 Watcher 注销 Host
├── 停内部事件线程
└── 置 Stopped（此后全部方法返回 TrayError::Stopped）
