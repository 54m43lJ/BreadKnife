1. 文件树（规划）

src/
├── lib.rs               # 公开面出口（核心 + gtk 整合层 + testing）
├── config.rs            # TrayConfig
├── handle.rs            # TrayHandle 前端（指令受理 → 内部线程投递）
├── events.rs            # TrayEvent、通道与回调分发
├── bus.rs               # 会话总线连接与 zbus 执行器宿主
├── host.rs              # Host 注册 + Item 发现/移除跟踪
├── props.rs             # Item 属性拉取 + PropertiesChanged 去抖
├── heartbeat.rs         # 周期存活探测
├── menu.rs              # dbusmenu 现场读取（零缓存）
├── interact.rs          # 交互转发 + InteractionResult 回执
├── watcher_track.rs     # Watcher 名字所有者跟踪
├── gtk/
│   ├── mod.rs           # 整合层出口（主线程约束说明）
│   ├── paintable.rs     # 渲染直出：主题命中 → 主题图标；未命中 → 位图纹理
│   └── bridge.rs        # 事件桥接：通道 → GTK 主线程（GLib 主循环）搬运
└── testing/
    ├── demo_item.rs     # 可编程演示 Item
    └── stub_watcher.rs  # 最小 Watcher 替身

运行约束：内部线程固定为 1（事件线程，同时宿主 zbus 执行器——DBus 收发、事件组装与投递在同一线程完成），不创建线程池。

2. 内部类型与方法签名

设计态骨架：描述各模块的内部组装方式与职责边界，实现落地后以实际代码为准。 公开类型与签名（Tray / TrayHandle / TrayEvent / 数据模型 / TrayError / gtk / testing）见 ARCHITECTURE.html §3，此处不重复。

// ── events.rs：事件出队后的分发枢纽（投递链 ①，ARCHITECTURE §2.2）──
pub(crate) struct EventHub {
    callbacks: Vec<Box<dyn Fn(&TrayEvent) + Send + 'static>>,  // 按注册顺序串行
    senders: Vec<mpsc::Sender<TrayEvent>>,   // connect / subscribe 追加的通道
    dropped: AtomicU64,                       // 背压丢弃告警计数
}
impl EventHub {
    pub(crate) fn register(&mut self, cb: Box<dyn Fn(&TrayEvent) + Send + 'static>);
    pub(crate) fn attach(&mut self) -> Receiver<TrayEvent>;
    /// 回调依次同步执行；逐通道 try_send，满则丢弃并递增 dropped
    pub(crate) fn dispatch(&self, ev: &TrayEvent);
    pub(crate) fn shutdown(&self);
}

// ── bus.rs：会话总线连接与 zbus 执行器宿主 ──
pub(crate) struct BusCenter {
    conn: zbus::Connection,                   // 唯一连接，宿主于内部事件线程
}
impl BusCenter {
    pub(crate) fn connect(timeout: Duration) -> Result<Self, TrayError>;
    /// 注册本端服务名（如 Host 服务名）；name_owner_changed 回调供跟踪方使用
    pub(crate) fn register_name(&self, name: &str) -> Result<(), TrayError>;
    pub(crate) fn watch_owner(&self, name: &str, on_change: impl Fn(Option<String>) + Send + 'static);
    pub(crate) fn call(&self, dest: &str, path: &str, iface: &str, method: &str,
                       body: &[zbus::zvariant::Value]) -> Result<Vec<zbus::zvariant::OwnedValue>, TrayError>;
}

// ── host.rs：Host 注册 + Item 发现/移除跟踪 ──
pub(crate) struct ItemEntry {
    item: TrayItem,
    menu_path: Option<String>,   // Menu 属性指向的 dbusmenu 对象路径，供 menu() 使用
    unresponsive: bool,
}
pub(crate) struct ItemTracker { items: HashMap<TrayItemId, ItemEntry> }
impl ItemTracker {
    /// RegisteredStatusNotifierItems 初始清单 → 逐个建立 Item 记录
    pub(crate) fn discover(&mut self, bus: &BusCenter) -> Result<Vec<TrayItem>, TrayError>;
    pub(crate) fn on_registered(&mut self, service: &str, bus: &BusCenter);
    pub(crate) fn on_unregistered(&mut self, service: &str);
    pub(crate) fn apply_fields(&mut self, id: &TrayItemId, fields: &[ItemField]) -> Option<TrayItem>;
}

// ── props.rs：属性拉取 + 去抖 ──
pub(crate) struct Debouncer {
    pending: HashMap<TrayItemId, BTreeSet<ItemField>>,
    window: Duration,            // TrayConfig::debounce_ms
}
impl Debouncer {
    pub(crate) fn on_field(&mut self, id: TrayItemId, f: ItemField);  // 刷新计时
    /// 窗口到期：返回各 Item 的字段并集（Changed 事件原料）
    pub(crate) fn flush(&mut self) -> Vec<(TrayItemId, Vec<ItemField>)>;
}
/// 拉取单个 Item 全部官方属性并按 ARCHITECTURE §3.3 字段表映射为 TrayItem（协议类失败按路线降级）
pub(crate) fn fetch_item(bus: &BusCenter, service: &str) -> Result<TrayItem, TrayError>;

// ── heartbeat.rs：周期存活探测 ──
pub(crate) struct Heartbeat { failures: HashMap<TrayItemId, u32> }
impl Heartbeat {
    /// 对每个在册 Item 做轻量属性读取（超时 timeout_ms），返回存亡变迁
    pub(crate) fn probe(&mut self, bus: &BusCenter, items: &ItemTracker) -> Vec<(TrayItemId, bool)>;
}

// ── menu.rs：dbusmenu 现场读取（零缓存）──

3. 关键流程调用栈（伪码）

3.1 启动（connect）

流程 connect(config)
├── 连接会话总线
├── 启动内部事件线程（zbus 执行器宿主）
├── 注册 Host 服务名（org.freedesktop.StatusNotifierHost-id，SNI §5）
├── 调用 Watcher.RegisterStatusNotifierHost
├── 探测 Watcher 存在性（IsStatusNotifierHostRegistered）→ 记录初始存亡
├── 查询在册 Item 清单（RegisteredStatusNotifierItems）
│    └── 对每个 Item
│         ├── 拉取全属性 → 建立 Item 记录
│         └── 订阅其 PropertiesChanged 与官方 New* 信号（SNI §3.3）
├── 事件入队 Snapshot(全部 Item, Watcher 存亡)
└── 返回 (Handle, 事件通道)

3.2 属性变更去抖

内部事件线程: 收到 PropertiesChanged(item, 字段)
             或官方 New* 信号 (NewStatus/NewIcon/NewTitle/
                               NewToolTip/NewOverlayIcon/NewAttentionIcon)
├── 追加进该 item 的待发变更集
├── 刷新去抖窗口计时
└── 窗口到期
     └── 合成 Changed{字段并集, 最新记录} → 入队 → 分发给回调与全部通道

3.3 菜单现场读取

流程 menu(id)
├── 校验 Item 在册（否则 ItemNotFound）
├── GetLayout（深度截断 menu_max_depth）
├── GetGroupProperties
├── 组装 MenuSnapshot
│    ├── 解析助记符 "_" → label
│    ├── 保留原文 → raw_label
│    └── 按 visible/enabled/toggled 归一化条目
└── 返回快照（不落任何缓存）

内部: 收到 LayoutUpdated / ItemsPropertiesUpdated / AboutToShow
└── 入队 MenuChanged(item)（不代为重拉）

3.4 交互转发

流程 activate(id, x, y)   （secondary_activate / scroll / menu_activate 同构）
├── 校验 Item 在册
├── 向 Item 发起 DBus 调用（受理即返回）
├── 同步返回 Ok / Err
└── 调用结果回到内部线程
     └── 入队 InteractionResult{op, 成功/失败}

3.5 心跳循环

内部事件线程: 周期心跳
├── 对每个在册 Item 做轻量属性读取（超时 timeout_ms）
├── 成功 → 清零失败计数；若曾标记 → 入队 Unresponsive(false)
└── 失败 → 累加失败计数
     └── 达到阈值 → 入队 Unresponsive(true)（不移除）

3.6 Watcher 跟踪与停机

内部: Watcher 名字所有者变化
├── 消失 → 入队 Narration(重连准备) + WatcherChanged(ExternalDown)
│         └── 重连成功后重新发现 Item（叙述 + 修正事件）
└── 出现 → 入队 WatcherChanged(ExternalUp)

流程 shutdown()
├── 向 Watcher 注销 Host
├── 停内部事件线程
└── 置 Stopped（此后全部方法返回 TrayError::Stopped）
